4) แยกตัวอักษร

Workflow แยกตัวอักษรจากภาพเอกสาร

อ่านภาพ → แปลงเทา → ลดนอยส์ → Threshold → Morphology → หา Contours → ครอปอักษร 
import cv2
import numpy as np

img = cv2.imread("images/text_sample.png", cv2.IMREAD_COLOR)
if img is None:
    raise FileNotFoundError("ไม่พบไฟล์ images/text_sample.png")

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 1) ลดนอยส์
gray_blur = cv2.medianBlur(gray, 3)

# 2) แยกพื้นหลัง/ตัวอักษร
# ใช้ THRESH_BINARY_INV เพราะต้องการให้ตัวอักษรเป็นสีขาวบนพื้นดำ
bin_img = cv2.adaptiveThreshold(
    gray_blur,
    255,
    cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
    cv2.THRESH_BINARY_INV,
    31,
    12
)

# 3) Morphology เพื่อเชื่อมส่วนที่ขาด และลดจุดรบกวน
kernel = np.ones((3, 3), np.uint8)
clean = cv2.morphologyEx(bin_img, cv2.MORPH_OPEN, kernel, iterations=1)
clean = cv2.morphologyEx(clean, cv2.MORPH_CLOSE, kernel, iterations=1)

# 4) หา contour ของตัวอักษร
contours, _ = cv2.findContours(clean, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

char_boxes = []
for cnt in contours:
    x, y, w, h = cv2.boundingRect(cnt)

    # กรองกล่องเล็กมาก (noise)
    if w < 5 or h < 8:
        continue

    char_boxes.append((x, y, w, h))

# เรียงซ้ายไปขวา (เหมาะกับบรรทัดเดียว)
char_boxes = sorted(char_boxes, key=lambda b: b[0])

# 5) ครอปและบันทึกตัวอักษร
for i, (x, y, w, h) in enumerate(char_boxes):
    pad = 2
    x1 = max(0, x - pad)
    y1 = max(0, y - pad)
    x2 = min(clean.shape[1], x + w + pad)
    y2 = min(clean.shape[0], y + h + pad)

    char_img = clean[y1:y2, x1:x2]
    cv2.imwrite(f"char_{i:03d}.png", char_img)

    # วาดกรอบบนภาพต้นฉบับเพื่อ debug
    cv2.rectangle(img, (x1, y1), (x2, y2), (0, 255, 0), 1)

cv2.imwrite("debug_boxes.png", img)
cv2.imwrite("debug_binary.png", clean)
print(f"เจอตัวอักษรทั้งหมด: {len(char_boxes)}")
5) ปรับให้แม่นขึ้น

เทคนิคเพิ่มคุณภาพ Segmentation

  • ปรับพารามิเตอร์ adaptiveThreshold เช่น blockSize, C
  • ถ้าภาพเอียง ใช้ deskew ก่อน threshold
  • แบ่งเป็นบรรทัดก่อน แล้วค่อยตัดเป็นตัวอักษร
  • เพิ่มเงื่อนไขกรองกล่องด้วยสัดส่วน w/h และพื้นที่
  • ถ้าเป็นไทย อาจเจอวรรณยุกต์ลอย ควรใช้กฎรวมกล่องที่อยู่ใกล้กัน
