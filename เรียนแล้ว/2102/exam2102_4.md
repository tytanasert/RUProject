```cpp
#include <iostream>
#include <cstring>
using namespace std;

// ==================== Date ====================
class Date {
private:
    int day, month, year;
public:
    // Default Constructor
    Date() : day(1), month(1), year(2000) {}

    // Parameterized Constructor
    Date(int d, int m, int y) : day(d), month(m), year(y) {}

    // Destructor
    ~Date() { cout << "~Date()" << endl; }

    // Getter / Setter
    void setDate(int d, int m, int y) { day = d; month = m; year = y; }
    void setDate(const Date& d) { day = d.day; month = d.month; year = d.year; }
    int getDay() { return day; }
    int getMonth() { return month; }
    int getYear() { return year; }

    // Show
    void show() {
        cout << day << "/" << month << "/" << year;
    }

    // สำหรับ operator<<
    void showDate(ostream& os) {
        os << day << "/" << month << "/" << year;
    }

    // สำหรับ operator>>
    void inDate(istream& is) {
        cout << "Day: "; is >> day;
        cout << "Month: "; is >> month;
        cout << "Year: "; is >> year;
    }
};

ostream& operator<<(ostream& os, Date& d) {
    d.showDate(os);
    return os;
}

istream& operator>>(istream& is, Date& d) {
    d.inDate(is);
    return is;
}

// ==================== Author ====================
class Author {
private:
    string name;
    char* nationality;  // dynamic memory!
public:
    Author() : name("Unknown") {
        nationality = new char[10];
        strcpy(nationality, "N/A");
    }

    Author(string n, const char* nat) : name(n) {
        nationality = new char[strlen(nat) + 1];
        strcpy(nationality, nat);
    }

    ~Author() {
        delete[] nationality;
        cout << "~Author()" << endl;
    }

    // Deep Copy Constructor
    Author(const Author& a) : name(a.name) {
        nationality = new char[strlen(a.nationality) + 1];
        strcpy(nationality, a.nationality);
    }

    // Deep Copy operator=
    Author& operator=(const Author& a) {
        if (this == &a) return *this;
        name = a.name;
        delete[] nationality;
        nationality = new char[strlen(a.nationality) + 1];
        strcpy(nationality, a.nationality);
        return *this;
    }

    // Getter / Setter
    void setName(string n) { name = n; }
    string getName() { return name; }
    void setNationality(const char* nat) {
        delete[] nationality;
        nationality = new char[strlen(nat) + 1];
        strcpy(nationality, nat);
    }
    char* getNationality() { return nationality; }

    void show() {
        cout << "Author: " << name << " (" << nationality << ")";
    }

    void showAuthor(ostream& os) {
        os << "Author: " << name << ", Nationality: " << nationality;
    }

    void inAuthor(istream& is) {
        cout << "Author Name: "; is >> name;
        delete[] nationality;
        nationality = new char[50];
        cout << "Nationality: "; is >> nationality;
    }
};

ostream& operator<<(ostream& os, Author& a) {
    a.showAuthor(os);
    return os;
}

istream& operator>>(istream& is, Author& a) {
    a.inAuthor(is);
    return is;
}

// ==================== Book ====================
class Book {
private:
    string title;
    Author author;       // Composition (Has-A)
    Date publishDate;    // Composition (Has-A)
    double* price;       // Dynamic memory!
public:
    Book() : title("Untitled") {
        price = new double(0);
    }

    Book(string t, string authName, const char* nat, int d, int m, int y, double p)
        : title(t), author(authName, nat), publishDate(d, m, y) {
        price = new double(p);
    }

    ~Book() {
        delete price;
        cout << "~Book()" << endl;
    }

    // Deep Copy Constructor
    Book(const Book& b) : title(b.title), author(b.author), publishDate(b.publishDate) {
        price = new double(*b.price);
    }

    // Deep Copy operator=
    Book& operator=(const Book& b) {
        if (this == &b) return *this;
        title = b.title;
        author = b.author;
        publishDate = b.publishDate;
        *price = *b.price;
        return *this;
    }

    // Getter / Setter
    void setTitle(string t) { title = t; }
    string getTitle() { return title; }
    void setPrice(double p) { *price = p; }
    double getPrice() { return *price; }

    void show() {
        cout << "Title: " << title << endl;
        author.show(); cout << endl;
        cout << "Published: "; publishDate.show(); cout << endl;
        cout << "Price: " << *price << endl;
    }

    void showBook(ostream& os) {
        os << "Title: " << title << endl;
        author.showAuthor(os); os << endl;
        os << "Published: "; publishDate.showDate(os); os << endl;
        os << "Price: " << *price << endl;
    }

    void inBook(istream& is) {
        cout << "Title: "; is >> title;
        author.inAuthor(is);
        publishDate.inDate(is);
        cout << "Price: "; is >> *price;
    }
};

ostream& operator<<(ostream& os, Book& b) {
    b.showBook(os);
    return os;
}

istream& operator>>(istream& is, Book& b) {
    b.inBook(is);
    return is;
}

// ==================== main ====================
int main() {
    Book b1("CPP_Primer", "Lippman", "American", 1, 1, 2020, 590.0);
    b1.show();
    cout << "---" << endl;

    Book b2(b1);     // Copy Constructor
    b2.show();

    Book b3;
    b3 = b1;         // operator=

    cin >> b3;        // operator>>
    cout << b3;       // operator<<

    return 0;
}
```

```cpp
#include <iostream>
#include <cstring>
using namespace std;

// ==================== Date ====================
class Date {
private:
    int day, month, year;
public:
    Date() : day(1), month(1), year(2000) {}
    Date(int d, int m, int y) : day(d), month(m), year(y) {}
    ~Date() { cout << "Date bye bye" << endl; }

    void setDate(Date& d) { day = d.day; month = d.month; year = d.year; }
    void setDate(int d, int m, int y) { day = d; month = m; year = y; }
    int getDay() { return day; }
    int getMonth() { return month; }
    int getYear() { return year; }

    void show() { cout << day << "/" << month << "/" << year << endl; }

    void inDate(istream& is) {
        cout << "day: "; is >> day;
        cout << "month: "; is >> month;
        cout << "year: "; is >> year;
    }
};

// ==================== Address ====================
class Address {
private:
    int houseNo;
    string street, city, state, zip;
public:
    Address() : houseNo(0), street(""), city(""), state(""), zip("") {}
    Address(int h, string st, string c, string s, string z)
        : houseNo(h), street(st), city(c), state(s), zip(z) {}
    ~Address() { cout << "Address bye bye" << endl; }

    void setAddress(Address& a) {
        houseNo = a.houseNo; street = a.street;
        city = a.city; state = a.state; zip = a.zip;
    }
    void setAddress(int h, string st, string c, string s, string z) {
        houseNo = h; street = st; city = c; state = s; zip = z;
    }

    int getHouseNo() { return houseNo; }
    string getStreet() { return street; }
    string getCity() { return city; }
    string getState() { return state; }
    string getZip() { return zip; }

    void show() {
        cout << houseNo << " " << street << ", " << city << ", " << state << " " << zip << endl;
    }

    void inAddress(istream& is) {
        cout << "houseNo: "; is >> houseNo;
        cout << "street: "; is >> street;
        cout << "city: "; is >> city;
        cout << "state: "; is >> state;
        cout << "zip: "; is >> zip;
    }
};

// ==================== Person ====================
class Person {
private:
    string fname, lname;
    Date date;           // Composition
    Address address;     // Composition
    string* status;      // Dynamic memory!
public:
    Person() : fname(""), lname("") {
        status = new string("");
    }

    Person(string fn, string ln, int d, int m, int y,
           int hNum, string st, string c, string s, string zip, string& sta)
        : fname(fn), lname(ln) {
        date.setDate(d, m, y);
        address.setAddress(hNum, st, c, s, zip);
        status = new string(sta);
    }

    Person(string fn, string ln, Date& d, Address& a, string& sta)
        : fname(fn), lname(ln), date(d), address(a) {
        status = new string(sta);
    }

    ~Person() {
        delete status;
        cout << "Person bye bye" << endl;
    }

    // ===== Deep Copy Constructor =====
    Person(const Person& p) : fname(p.fname), lname(p.lname),
                               date(p.date), address(p.address) {
        status = new string(*p.status);  // Deep copy!
    }

    // ===== Deep Copy operator= =====
    Person& operator=(Person& p) {
        if (this == &p) return *this;   // Self-assignment check
        fname = p.fname;
        lname = p.lname;
        date = p.date;
        address = p.address;
        *status = *p.status;           // Deep copy ค่า
        return *this;
    }

    // Getter / Setter
    void setFname(string fn) { fname = fn; }
    void setLname(string ln) { lname = ln; }
    void setDate(Date& d) { date = d; }
    void setDate(int d, int m, int y) { date = Date(d, m, y); }
    void setAddress(Address a) { address = a; }
    void setStatus(string& sta) { *status = sta; }

    string getFname() { return fname; }
    string getLname() { return lname; }
    Date getDate() { return date; }
    Address getAddress() { return address; }
    string getStatus() { return *status; }

    void show() {
        cout << "First Name: " << fname << endl;
        cout << "Last Name: " << lname << endl;
        cout << "Status: " << *status << endl;
        date.show();
        address.show();
    }

    void showPerson(ostream& os) {
        os << "fname: " << fname << " lname: " << lname << endl;
        os << "status: " << *status << endl;
        address.show();
        date.show();
    }

    void setPerson(istream& is) {
        cout << "fname: "; is >> fname;
        cout << "lname: "; is >> lname;
        address.inAddress(is);
        date.inDate(is);
        cout << "status: "; is >> *status;
    }
};

ostream& operator<<(ostream& os, Person& p) { p.showPerson(os); return os; }
istream& operator>>(istream& is, Person& p) { p.setPerson(is); return is; }

// ==================== Student (Virtual Inheritance) ====================
class Student : virtual public Person {
    string studentID;
    string major;
    int* year;           // Dynamic memory!
public:
    Student() : studentID(""), major("") {
        year = new int(2569);
    }

    Student(string fn, string ln, int d, int m, int y,
            int hNum, string st, string c, string s, string zip,
            string sid, string maj, string sta, int& yea)
        : Person(fn, ln, d, m, y, hNum, st, c, s, zip, sta),
          studentID(sid), major(maj) {
        year = new int(yea);
    }

    ~Student() {
        delete year;
        cout << "Student bye bye" << endl;
    }

    // ===== Deep Copy Constructor =====
    Student(Student& p) : Person(p) {
        studentID = p.studentID;
        major = p.major;
        year = new int(*p.year);   // Deep copy!
    }

    // ===== Deep Copy operator= =====
    Student& operator=(Student& p) {
        if (this == &p) return *this;
        Person::operator=(p);      // เรียก operator= ของ Person
        studentID = p.studentID;
        major = p.major;
        *year = *p.year;           // Deep copy ค่า
        return *this;
    }

    // Getter / Setter
    void setStudentId(string sid) { studentID = sid; }
    string getStudentId() { return studentID; }
    void setMajor(string m) { major = m; }
    string getMajor() { return major; }
    void setYear(int& y) { *year = y; }
    int* getYear() { return year; }

    void show() {
        Person::show();
        cout << "Student ID: " << studentID << endl;
        cout << "Major: " << major << endl;
        cout << "Year: " << *year << endl;
    }

    void showStudent(ostream& os) {
        Person::showPerson(os);
        os << "studentID: " << studentID << " major: " << major << endl;
        os << "year: " << *year << endl;
    }

    void setStudent(istream& is) {
        Person::setPerson(is);
        cout << "studentID: "; is >> studentID;
        cout << "major: "; is >> major;
        cout << "year: "; is >> *year;
    }
};

ostream& operator<<(ostream& os, Student& s) { s.showStudent(os); return os; }
istream& operator>>(istream& is, Student& s) { s.setStudent(is); return is; }

// ==================== Teacher (Virtual Inheritance) ====================
class Teacher : virtual public Person {
    string teacherID;
    string subject;
    char* position;      // Dynamic memory (char array)!
public:
    Teacher() : teacherID(""), subject("") {
        position = new char[20];
        strcpy(position, "dr");
    }

    Teacher(string fn, string ln, int d, int m, int y,
            int hNum, string st, string c, string s, string zip,
            string tid, string sub, const char pos[], string& sta)
        : Person(fn, ln, d, m, y, hNum, st, c, s, zip, sta),
          teacherID(tid), subject(sub) {
        position = new char[strlen(pos) + 1];
        strcpy(position, pos);
    }

    ~Teacher() {
        delete[] position;
        cout << "Teacher bye bye" << endl;
    }

    // ===== Deep Copy Constructor =====
    Teacher(Teacher& t) : Person(t) {
        teacherID = t.teacherID;
        subject = t.subject;
        position = new char[strlen(t.position) + 1];  // จอง memory ใหม่
        strcpy(position, t.position);                  // คัดลอกข้อมูล
    }

    // ===== Deep Copy operator= =====
    Teacher& operator=(Teacher& t) {
        if (this == &t) return *this;
        Person::operator=(t);
        teacherID = t.teacherID;
        subject = t.subject;
        delete[] position;                              // ลบของเดิม
        position = new char[strlen(t.position) + 1];    // จอง memory ใหม่
        strcpy(position, t.position);                   // คัดลอกข้อมูล
        return *this;
    }

    // Getter / Setter
    void setTeacherId(string tid) { teacherID = tid; }
    string getTeacherId() { return teacherID; }
    void setSubject(string sub) { subject = sub; }
    string getSubject() { return subject; }
    void setPosition(char* pos) {
        delete[] position;
        position = new char[strlen(pos) + 1];
        strcpy(position, pos);
    }
    char* getPosition() { return position; }

    void show() {
        Person::show();
        cout << "Teacher ID: " << teacherID << endl;
        cout << "Subject: " << subject << endl;
        cout << "Position: " << position << endl;
    }

    void showTeacher(ostream& os) {

    void setTeacher(istream& is) {

        delete[] position;
        position = new char[50];
        cout << "Position: "; is >> position;

};

ostream& operator<<(ostream& os, Teacher& t) { t.showTeacher(os); return os; }
istream& operator>>(istream& is, Teacher& t) { t.setTeacher(is); return is; }

// ========== TeachingAssistant (Multiple + Virtual Inheritance) ==========
class TeachingAssistant : public Student, public Teacher {
private:
    string taID;
    double* salary;      // Dynamic memory!
public:
    TeachingAssistant() {
        salary = new double(1000);
    }

    TeachingAssistant(string fn, string ln, int d, int m, int y,
                      int hNum, string st, string c, string s, string zip, string sta,
                      string sid, string maj, int yea,
                      string tid, string sub, char* pos,
                      string taId, double& sal)
        : Person(fn, ln, d, m, y, hNum, st, c, s, zip, sta)  // ← เรียก Person ตรงๆ!
        , Student(fn, ln, d, m, y, hNum, st, c, s, zip, sid, maj, sta, yea)
        , Teacher(tid, sub, pos) {
        taID = taId;
        salary = new double(sal);
    }

    ~TeachingAssistant() {
        delete salary;
        cout << "TA bye bye" << endl;
    }

    // ===== Deep Copy Constructor =====
    TeachingAssistant(TeachingAssistant& ta) : Person(ta), Student(ta), Teacher(ta) {
        taID = ta.taID;
        salary = new double(*ta.salary);  // Deep copy!
    }

    // ===== Deep Copy operator= =====
    TeachingAssistant& operator=(TeachingAssistant& ta) {
        if (this == &ta) return *this;
        Person::operator=(ta);
        Student::operator=(ta);
        Teacher::operator=(ta);
        taID = ta.taID;
        *salary = *ta.salary;             // Deep copy ค่า
        return *this;
    }

    // Getter / Setter
    void setTaId(string tid) { taID = tid; }
    string getTaId() { return taID; }
    void setSalary(double& sal) { *salary = sal; }
    double getSalary() { return *salary; }

    void show() {
        Person::show();
        Student::show();
        Teacher::show();
        cout << "TA ID: " << taID << endl;
        cout << "Salary: " << *salary << endl;

        Student::showStudent(os);
        Teacher::showTeacher(os);
        os << "TA Id: " << taID << endl;
        os << "Salary: " << *salary << endl;
    }

    void setTeachingAssistant(istream& is) {
        Person::setPerson(is);
        Student::setStudent(is);
        Teacher::setTeacher(is);
        cout << "TA Id: "; is >> taID;
        delete salary;
        salary = new double();
        cout << "Salary: "; is >> *salary;
    }
};

ostream& operator<<(ostream& os, TeachingAssistant& ta) {
    ta.showTeachingAssistant(os);
    return os;
}

istream& operator>>(istream& is, TeachingAssistant& ta) {
    ta.setTeachingAssistant(is);
    return is;
}
```

## 📘 Final 77 — University System (ครบทุก concept)

[🔝 กลับสารบัญ](#📑-สารบัญ)

### โจทย์แนว
โจทย์แบบเดียวกับ Exam8.cpp ที่ให้มา:
- `Date`, `Address` — class พื้นฐาน
- `Person` — มี Date, Address [composition], string* status [dynamic]
- `Student : virtual public Person` — มี studentID, major, int* year [dynamic]
- `Teacher : virtual public Person` — มี teacherID, subject, char* position [dynamic]
- `TeachingAssistant : public Student, public Teacher` — มี taID, double* salary [dynamic]

เน้น:
- Virtual Inheritance (แก้ Diamond Problem)
- Deep Copy (Copy Constructor + operator=)
- Operator Overloading (<< >>)
- ลำดับ Constructor / Destructor

### เฉลย

> ดูโค้ดเต็มได้ที่ **[Final 58](#final-58)** ซึ่งเป็นโจทย์รูปแบบเดียวกัน
> ด้านล่างนี้จะเน้นจุดที่แตกต่างหรือต้องระวังเป็นพิเศษ

### จุดสำคัญที่ต้องระวัง

**1. Virtual Inheritance → TA ต้องเรียก Person constructor ตรงๆ**
```cpp
class TeachingAssistant : public Student, public Teacher {
    // Constructor ต้องเรียก Person ตรง!
    TeachingAssistant(/*params*/)
        : Person(fn, ln, d, m, y, hNum, st, c, s, zip, sta)  // ← ต้องมี!
        , Student(/*...*/)
        , Teacher(/*...*/) {
        // ...
    }
};
```

**2. Copy Constructor ของ TA ต้องเรียกทั้ง Person, Student, Teacher**
```cpp
TeachingAssistant(TeachingAssistant& ta)
    : Person(ta)      // ← Virtual base
    , Student(ta)
    , Teacher(ta) {
    taID = ta.taID;
    salary = new double(*ta.salary);  // Deep copy!
}
```

**3. operator= ของ TA ต้องเรียก operator= ของทุก base**
```cpp
TeachingAssistant& operator=(TeachingAssistant& ta) {
    if (this == &ta) return *this;
    Person::operator=(ta);     // ← เรียก Person operator=
    Student::operator=(ta);    // ← เรียก Student operator=
    Teacher::operator=(ta);    // ← เรียก Teacher operator=
    taID = ta.taID;
    *salary = *ta.salary;
    return *this;
}
```

**4. ลำดับ Constructor ของ TA**
```
Date()        ← member ของ Person (composition)
Address()     ← member ของ Person (composition)
Person()      ← Virtual base (สร้างก่อนเสมอ, ครั้งเดียว)
Student()     ← Base class ตัวแรก (ตาม declaration order)
Teacher()     ← Base class ตัวที่สอง
TA()          ← ตัวเอง
```

**5. ลำดับ Destructor ของ TA (ย้อนกลับ)**
```
~TA()
~Teacher()
~Student()
~Person()
~Address()
~Date()
```

**6. char* ต้องใช้ strcpy ไม่ใช่ = (assignment)**
```cpp
// ❌ ผิด!
position = t.position;  // Shallow copy! ชี้ที่เดียวกัน

// ✅ ถูก!
position = new char[strlen(t.position) + 1];
strcpy(position, t.position);
```

**7. delete vs delete[]**
```cpp
int* year;       → delete year;      // ใช้ delete ธรรมดา
char* position;  → delete[] position; // ใช้ delete[] สำหรับ array
string* status;  → delete status;     // ใช้ delete ธรรมดา
double* salary;  → delete salary;     // ใช้ delete ธรรมดา
```
